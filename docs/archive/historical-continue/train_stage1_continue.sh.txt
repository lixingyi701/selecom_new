#!/bin/bash
# Continue encoder full fine-tuning after Stage 2, using Stage 1 data + selection objective.
# Low LR (5e-6), 5000 samples, 1 epoch — diagnostic probe run.
export CUDA_VISIBLE_DEVICES=2,3
export MASTER_PORT=55557
export HF_HUB_OFFLINE=1
export TRANSFORMERS_OFFLINE=1

ENCODER_NAME="../baselineModel/Qwen3-Embedding-0.6B"
GENERATOR_NAME="../baselineModel/Qwen2.5-7B-Instruct"
STAGE1_CHECKPOINT_DIR="../checkpoint/pretrainedModel/Qwen3embedding0.6B-Qwen2.57B-selector"
STAGE2_CHECKPOINT_DIR="../checkpoint/pretrainedModel/Qwen3embedding0.6B-Qwen2.57B-generator"
DATA_PATH="../data/stage1/stage1_train_data.jsonl"
OUTPUT_DIR="../results/stage1_continue"
LOG_DIR="../results/stage1_continue/logs"

NUM_GPUS=$(echo $CUDA_VISIBLE_DEVICES | awk -F',' '{print NF}')
mkdir -p "$OUTPUT_DIR" "$LOG_DIR"

nohup torchrun --nproc_per_node=$NUM_GPUS --master_port=$MASTER_PORT train_stage1_continue.py \
    --encoder_name "$ENCODER_NAME" \
    --generator_name "$GENERATOR_NAME" \
    --stage1_checkpoint_dir "$STAGE1_CHECKPOINT_DIR" \
    --stage2_checkpoint_dir "$STAGE2_CHECKPOINT_DIR" \
    --data_path "$DATA_PATH" \
    --max_samples 5000 \
    --model_dir "$OUTPUT_DIR" \
    --log_dir "$LOG_DIR" \
    --log_name "train_stage1_continue.log" \
    --epochs 1 \
    --learning_rate 5e-6 \
    --batch_size 8 \
    --gradient_accumulation_steps 4 \
    --logging_steps 20 \
    >> "$LOG_DIR/train_stage1_continue.log" 2>&1 &

echo "Training started (PID $!). Log: $LOG_DIR/train_stage1_continue.log"
# bash train_stage1_continue.sh
